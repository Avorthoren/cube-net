For xcb to work these have to be installed:
```
sudo apt-get update

sudo apt-get install -y \
    libxcb-cursor0 \
    libxcb-xinerama0 \
    libxcb-image0 \
    libxcb-icccm4 \
    libxcb-randr0 \
    libxcb-sync1 \
    libxcb-xkb1 \
    libxcb-render-util0 \
    libxcb-keysyms1 \
    libxcb-shape0 \
    libx11-xcb1 \
    libxkbcommon-x11-0
```
